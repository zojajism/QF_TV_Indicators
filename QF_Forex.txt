//@version=6
indicator("HT1 / HT2 Previous Range Boxes - Historical", overlay = true, max_boxes_count = 500, max_labels_count = 500)

//---------------------------------------------------------
// Inputs
//---------------------------------------------------------
groupHT1 = "HT1 - Higher Higher Timeframe"
ht1        = input.timeframe("D", "HT1 Timeframe", group = groupHT1)
showHT1    = input.bool(true, "Show HT1 Boxes", group = groupHT1)
ht1Border  = input.color(color.new(color.blue, 0), "HT1 Border Color", group = groupHT1)
ht1Fill    = input.color(color.new(color.blue, 85), "HT1 Fill Color", group = groupHT1)
ht1Width   = input.int(1, "HT1 Border Width", minval = 1, maxval = 5, group = groupHT1)
ht1History = input.int(100, "HT1 Max Boxes", minval = 1, maxval = 250, group = groupHT1)

groupHT2 = "HT2 - Higher Timeframe"
ht2        = input.timeframe("240", "HT2 Timeframe", group = groupHT2)
showHT2    = input.bool(true, "Show HT2 Boxes", group = groupHT2)
ht2Border  = input.color(color.new(color.orange, 0), "HT2 Border Color", group = groupHT2)
ht2Fill    = input.color(color.new(color.orange, 85), "HT2 Fill Color", group = groupHT2)
ht2Width   = input.int(1, "HT2 Border Width", minval = 1, maxval = 5, group = groupHT2)
ht2History = input.int(100, "HT2 Max Boxes", minval = 1, maxval = 250, group = groupHT2)

groupDaily = "Custom Daily Logic"
useCustomDaily = input.bool(true, "Use Custom Daily Logic Only When HTF = D", group = groupDaily, tooltip = "If enabled, any selected HTF equal to D will use the manual daily start hour instead of the exchange-defined daily candle.")
dailyStartHour = input.int(17, "Daily Start Hour", minval = 0, maxval = 23, group = groupDaily, tooltip = "Hour at which a new custom day begins. Default 17 matches the common Forex New York close.")

groupBias = "Bias Labels"
showBiasLabels = input.bool(true, "Show Bias Labels", group = groupBias)
bullBiasColor  = input.color(color.new(color.green, 0), "Bull Label Color", group = groupBias)
bearBiasColor  = input.color(color.new(color.red, 0), "Bear Label Color", group = groupBias)
neutralBiasColor = input.color(color.new(color.gray, 0), "Neutral Label Color", group = groupBias)
biasLabelSize  = input.string("Tiny", "Bias Label Size", options = ["Tiny", "Small", "Normal"], group = groupBias)

groupGen = "General"
showLabels = input.bool(true, "Show Box Labels", group = groupGen)
labelSize  = input.string("Tiny", "Box Label Size", options = ["Tiny", "Small", "Normal"], group = groupGen)

//---------------------------------------------------------
// Helpers
//---------------------------------------------------------
f_text_size(_size) =>
    _size == "Small" ? size.small : _size == "Normal" ? size.normal : size.tiny

f_validate_htf(_tf) =>
    timeframe.in_seconds(_tf) > timeframe.in_seconds(timeframe.period)

f_is_daily_tf(_tf) =>
    timeframe.in_seconds(_tf) == timeframe.in_seconds("D")

f_trim_boxes(_arr, _maxCount) =>
    while array.size(_arr) > _maxCount
        oldBox = array.shift(_arr)
        box.delete(oldBox)

f_trim_labels(_arr, _maxCount) =>
    while array.size(_arr) > _maxCount
        oldLbl = array.shift(_arr)
        label.delete(oldLbl)

f_bias_text(_open, _close) =>
    _close > _open ? "BULL" : _close < _open ? "BEAR" : "NEUTRAL"

f_bias_color(_open, _close, _bull, _bear, _neutral) =>
    _close > _open ? _bull : _close < _open ? _bear : _neutral

//---------------------------------------------------------
// Validation
//---------------------------------------------------------
if showHT1 and not f_validate_htf(ht1)
    runtime.error("HT1 must be higher than the current chart timeframe.")

if showHT2 and not f_validate_htf(ht2)
    runtime.error("HT2 must be higher than the current chart timeframe.")

//---------------------------------------------------------
// HTF slot detection
//---------------------------------------------------------
ht1Boundary = timeframe.change(ht1)
ht2Boundary = timeframe.change(ht2)

ht1CurrStart = time(ht1)
ht1CurrEnd   = time_close(ht1)

ht2CurrStart = time(ht2)
ht2CurrEnd   = time_close(ht2)

//---------------------------------------------------------
// Custom daily slot detection
//---------------------------------------------------------
msInDay = 24 * 60 * 60 * 1000
prevBarTime = nz(time[1], time)

todayAnchor = timestamp(syminfo.timezone, year(time), month(time), dayofmonth(time), dailyStartHour, 0)
prevTodayAnchor = timestamp(syminfo.timezone, year(prevBarTime), month(prevBarTime), dayofmonth(prevBarTime), dailyStartHour, 0)

currentCustomStart = time >= todayAnchor ? todayAnchor : todayAnchor - msInDay
prevCustomStart    = prevBarTime >= prevTodayAnchor ? prevTodayAnchor : prevTodayAnchor - msInDay
currentCustomEnd   = currentCustomStart + msInDay
newCustomDay       = currentCustomStart != prevCustomStart

//---------------------------------------------------------
// HT1 state
//---------------------------------------------------------
var float ht1RunHigh = na
var float ht1RunLow = na
var float ht1RunOpen = na
var float ht1RunClose = na
var array<box> ht1Boxes = array.new<box>()
var array<label> ht1BiasLabels = array.new<label>()

ht1UseCustomDaily = useCustomDaily and f_is_daily_tf(ht1)
ht1NewSlot        = ht1UseCustomDaily ? newCustomDay : ht1Boundary
ht1SlotStart      = ht1UseCustomDaily ? currentCustomStart : ht1CurrStart
ht1SlotEnd        = ht1UseCustomDaily ? currentCustomEnd : ht1CurrEnd
ht1Text           = ht1UseCustomDaily ? "HT1 Prev Custom D" : "HT1 Prev " + ht1

if barstate.isfirst
    ht1RunHigh := high
    ht1RunLow := low
    ht1RunOpen := open
    ht1RunClose := close

if ht1NewSlot
    if showHT1 and not na(ht1RunHigh) and not na(ht1RunLow) and not na(ht1SlotStart) and not na(ht1SlotEnd)
        ht1Box = box.new(
             left = ht1SlotStart,
             right = ht1SlotEnd,
             top = ht1RunHigh,
             bottom = ht1RunLow,
             xloc = xloc.bar_time,
             border_color = ht1Border,
             border_width = ht1Width,
             bgcolor = ht1Fill,
             text = showLabels ? ht1Text : "",
             text_color = ht1Border,
             text_size = f_text_size(labelSize),
             text_halign = text.align_left,
             text_valign = text.align_top
         )
        array.push(ht1Boxes, ht1Box)
        f_trim_boxes(ht1Boxes, ht1History)

        if showBiasLabels
            ht1BiasText = f_bias_text(ht1RunOpen, ht1RunClose)
            ht1BiasCol = f_bias_color(ht1RunOpen, ht1RunClose, bullBiasColor, bearBiasColor, neutralBiasColor)
            ht1BiasLabel = label.new(
                 x = ht1SlotStart,
                 y = ht1RunHigh,
                 text = ht1BiasText,
                 xloc = xloc.bar_time,
                 yloc = yloc.price,
                 style = label.style_label_lower_left,
                 color = ht1BiasCol,
                 textcolor = color.white,
                 size = f_text_size(biasLabelSize)
             )
            array.push(ht1BiasLabels, ht1BiasLabel)
            f_trim_labels(ht1BiasLabels, ht1History)

    ht1RunHigh := high
    ht1RunLow := low
    ht1RunOpen := open
    ht1RunClose := close
else
    ht1RunHigh := na(ht1RunHigh) ? high : math.max(ht1RunHigh, high)
    ht1RunLow := na(ht1RunLow) ? low : math.min(ht1RunLow, low)
    ht1RunOpen := na(ht1RunOpen) ? open : ht1RunOpen
    ht1RunClose := close

//---------------------------------------------------------
// HT2 state
//---------------------------------------------------------
var float ht2RunHigh = na
var float ht2RunLow = na
var float ht2RunOpen = na
var float ht2RunClose = na
var array<box> ht2Boxes = array.new<box>()
var array<label> ht2BiasLabels = array.new<label>()

ht2UseCustomDaily = useCustomDaily and f_is_daily_tf(ht2)
ht2NewSlot        = ht2UseCustomDaily ? newCustomDay : ht2Boundary
ht2SlotStart      = ht2UseCustomDaily ? currentCustomStart : ht2CurrStart
ht2SlotEnd        = ht2UseCustomDaily ? currentCustomEnd : ht2CurrEnd
ht2Text           = ht2UseCustomDaily ? "HT2 Prev Custom D" : "HT2 Prev " + ht2

if barstate.isfirst
    ht2RunHigh := high
    ht2RunLow := low
    ht2RunOpen := open
    ht2RunClose := close

if ht2NewSlot
    if showHT2 and not na(ht2RunHigh) and not na(ht2RunLow) and not na(ht2SlotStart) and not na(ht2SlotEnd)
        ht2Box = box.new(
             left = ht2SlotStart,
             right = ht2SlotEnd,
             top = ht2RunHigh,
             bottom = ht2RunLow,
             xloc = xloc.bar_time,
             border_color = ht2Border,
             border_width = ht2Width,
             bgcolor = ht2Fill,
             text = showLabels ? ht2Text : "",
             text_color = ht2Border,
             text_size = f_text_size(labelSize),
             text_halign = text.align_left,
             text_valign = text.align_top
         )
        array.push(ht2Boxes, ht2Box)
        f_trim_boxes(ht2Boxes, ht2History)

        if showBiasLabels
            ht2BiasText = f_bias_text(ht2RunOpen, ht2RunClose)
            ht2BiasCol = f_bias_color(ht2RunOpen, ht2RunClose, bullBiasColor, bearBiasColor, neutralBiasColor)
            ht2BiasLabel = label.new(
                 x = ht2SlotStart,
                 y = ht2RunHigh,
                 text = ht2BiasText,
                 xloc = xloc.bar_time,
                 yloc = yloc.price,
                 style = label.style_label_lower_left,
                 color = ht2BiasCol,
                 textcolor = color.white,
                 size = f_text_size(biasLabelSize)
             )
            array.push(ht2BiasLabels, ht2BiasLabel)
            f_trim_labels(ht2BiasLabels, ht2History)

    ht2RunHigh := high
    ht2RunLow := low
    ht2RunOpen := open
    ht2RunClose := close
else
    ht2RunHigh := na(ht2RunHigh) ? high : math.max(ht2RunHigh, high)
    ht2RunLow := na(ht2RunLow) ? low : math.min(ht2RunLow, low)
    ht2RunOpen := na(ht2RunOpen) ? open : ht2RunOpen
    ht2RunClose := close
